cmake_minimum_required(VERSION 3.22)
project(earthsim LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

option(ES_EXTENSION "build the Godot extension (needs godot-cpp)" ON)
option(ES_TOOLS "build tests and dev tools" ON)

# ---- core: pure C++, no Godot dependency
file(GLOB CORE_SRC ${CMAKE_CURRENT_SOURCE_DIR}/core/src/*.cpp)
add_library(escore STATIC ${CORE_SRC})
target_include_directories(escore PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/core/src)
if(MSVC)
  target_compile_options(escore PRIVATE /O2 /EHsc /bigobj /utf-8)
else()
  target_compile_options(escore PRIVATE -O2 -Wall -Wno-unused-function -Wno-sign-compare)
endif()

if(ES_TOOLS)
  add_executable(es_tests core/tests/tests.cpp)
  target_link_libraries(es_tests escore)
  foreach(t mvt_inspect region_info world_dump)
    add_executable(${t} tools/${t}.cpp)
    target_link_libraries(${t} escore)
  endforeach()
endif()

# ---- Godot extension
if(ES_EXTENSION)
  if(NOT GODOT_CPP_DIR)
    set(GODOT_CPP_DIR ${CMAKE_CURRENT_SOURCE_DIR}/godot-cpp)
  endif()
  add_subdirectory(${GODOT_CPP_DIR} godot-cpp EXCLUDE_FROM_ALL)
  file(GLOB EXT_SRC ${CMAKE_CURRENT_SOURCE_DIR}/ext/src/*.cpp)
  add_library(earthsim SHARED ${EXT_SRC})
  target_link_libraries(earthsim PRIVATE escore godot-cpp)
  if(MSVC)
    target_compile_options(earthsim PRIVATE /O2 /EHsc /utf-8)
  else()
    target_compile_options(earthsim PRIVATE -O2)
  endif()
  set_target_properties(earthsim PROPERTIES PREFIX "lib")
  set(OUT ${CMAKE_CURRENT_SOURCE_DIR}/game/bin)
  set_target_properties(earthsim PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${OUT} RUNTIME_OUTPUT_DIRECTORY ${OUT})
  foreach(cfg DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(earthsim PROPERTIES LIBRARY_OUTPUT_DIRECTORY_${cfg} ${OUT} RUNTIME_OUTPUT_DIRECTORY_${cfg} ${OUT})
  endforeach()
endif()
