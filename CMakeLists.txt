cmake_minimum_required(VERSION 3.16)
project(truetorque CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_library(ttcore STATIC unreal/Source/TrueTorque/Sim/softbody.cpp unreal/Source/TrueTorque/Sim/vehicle.cpp unreal/Source/TrueTorque/Sim/net.cpp)
target_include_directories(ttcore PUBLIC unreal/Source/TrueTorque/Sim)
add_executable(tt_tests tests/tests.cpp)
target_link_libraries(tt_tests ttcore)
