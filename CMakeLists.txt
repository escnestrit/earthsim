cmake_minimum_required(VERSION 3.16)
project(truetorque CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
# car geometry and visual damage (used by the game)
add_library(ttgeom STATIC unreal/Source/TrueTorque/Geom/ttgeom.cpp)
target_include_directories(ttgeom PUBLIC unreal/Source/TrueTorque/Geom)
add_executable(geom_tests tests/geom_tests.cpp)
target_link_libraries(geom_tests ttgeom)
add_executable(geom_preview tests/geom_preview.cpp)
target_link_libraries(geom_preview ttgeom)
# the earlier soft-body prototype (no longer in the game)
add_library(ttcore STATIC legacy_softbody/softbody.cpp legacy_softbody/vehicle.cpp legacy_softbody/net.cpp)
target_include_directories(ttcore PUBLIC legacy_softbody)
add_executable(tt_tests tests/tests.cpp)
target_link_libraries(tt_tests ttcore)
